# VFS загружается из --vfs до выполнения этого скрипта.
# Имя загруженной VFS видно в заголовке окна.

# ls показывает содержимое VFS, cd меняет текущую папку.
ls
ls -l .  # комментарий после команды
cd "/"
cd '.'

# exit с аргументом сообщает об ошибке и останавливает скрипт.
# Успешный exit показан отдельно в scripts/with_exit.txt.
exit 1
