# Стартовый скрипт без ошибок: выполняются все команды.

# Без кавычек: два отдельных аргумента
ls 123 123
# В кавычках: один аргумент с пробелом
ls "123 123"
ls -l 12 4 dv  # комментарий в конце строки отбрасывается
# Символ # внутри кавычек комментарием не считается
ls "dir#1"

# Команды без аргументов
ls
